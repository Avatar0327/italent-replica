## 0. 一页摘要

- **声明（§2）**：每条路由改用 `declare(router, method, path, policy, handler)` 注册。policy 十类：`public / platform / member / admin / self / own / relation / exception / button / object`，可用 `any`（OR）/ `all`（AND）组合；所有 policy 共用 `invalidId`、`guards`、`write` 等基础字段。`object` 的对象、操作、按钮都可以是**选择器**，每个动态选择器与逐行关系都登记**有限分支清单**；批量 / 导入用 `rows` 逐行声明操作、按钮、字段、范围目标与关系。范围外与标识非法都声明 **status + error.code**（DEC-291 Q1）。附录 A 对运行时枚举出的 **227 个端点**逐条给出确定映射，并与类型闭合（类型能承载附录承诺的全部能力）。
- **执行点（§3）**：最小安全解析 → 功能权限 → 写字段 → 范围 / 关系 → 处理函数（动作前业务前提在命令内、提交前足迹由框架钩子）→ 返回后（首次与重放都只做"当前授权 + 结果守卫 + 投影"，不重跑业务前提）→ 出口。写入声明必填 `footprint` 与 `result` 两个具名守卫，附录里每条写路由都给出具名守卫或"不适用 + 依据"。列表与审计走**框架组装与执行的查询契约**：行集合部分（FROM / INNER JOIN / CTE）禁止分页与聚合关键字，分页与聚合只由框架在范围过滤之后添加。出口按**响应路径 × 对象 × 编码**投影，字段权限取**实际通过的授权分支**（DEC-102 流程管理员整对象可见）。四个导入入口从任务构造起整体包在 `withFailedImportLog` 内，步骤 0 的可记录解析失败也进包。
- **缺失即失败（§2.3）**：按注册实例绑定最终 method / path；中间件按 **Hono 规范化并挂载后的最终路径**登记（`'*'` → `/*`，嵌套 `'*'` → `/api/tenant/approval/*`，实测）；别名、`.all`、未登记中间件、多层 `onError` 都能检出；用 Hono "匹配器已构建"不变量封闭注册。运行时自检按**有效匹配方法**（HEAD 继承 GET 的权限与无响应体语义）。
- **CI 反向测试（§4）**：八个固定用户只是**模板**（A / H / N / O / F / B / P / S），O / O_all 用独立身份克隆并在请求前断言实际范围；必测项由声明推导成"路由 × 权限分支 × 场景"矩阵，动态分支按登记清单做**集合相等**核对；负例用合法请求隔离单个权限；F 的写探针分"请求字段"与"继承 / 派生输出字段"两类；校验和分业务（含权限管理）/ 平台 / 台账（内容）/ 审计与出站四组。
- **分步上线（§3.7、§8）**：PR-A 迁移 **126** 个端点，**101** 个进精确 method / path 基线；基线与例外的**可信比较基准**是 `origin/main` 上的同名文件与 DEC 登记（CI 在 PR 上与 main 做子集比较），不是 PR 自己的 JSON；基线与例外同时为空即切强制。

## 2. 设计一：路由权限声明 v3

### 2.1 类型（`apps/api/src/route-policy/types.ts`）

```ts
/** 选择器：静态值，或由请求某处决定。取值只用步骤 0 的最小安全解析结果；取不到或不在映射表 / 分支清单 → 400 VALIDATION_FAILED，不进入鉴权。 */
type Selector<T> =
  | T
  | { from: 'param' | 'body' | 'query'; path: string; map: Readonly<Record<string, T>> } // 分支清单 = map 的键
  | { from: 'mapper'; mapper: MapperName; domain: readonly string[] } // 纯函数映射 + 有限输出域（P2-8）；域外输出 → 400
  | { from: 'record'; locator: LocatorName; attribute: string; domain: readonly string[] }; // 由已加载记录决定（审批快照 fieldObjectCode），域外 → 500 fail-closed

type ObjectSelector = Selector<ObjectCode>;
/** 'button' = 只校验按钮（按钮判定本身要求对象在身份清单内），不附加数据操作开关——审批管理员动作、流程仿真等现状。 */
type OperationSelector = Selector<'view' | 'create' | 'update' | 'delete' | 'button'>;
type ButtonPolicy = Selector<{ code: string; level: ButtonLevel }> | { none: true; reason: string };

/** 拒绝码：范围外 / 关系不成立 / 标识非法都声明 status 与 error.code（DEC-291 Q1），含 details.reason。 */
type Denial = { status: 400 | 403 | 404 | 409; code: ErrorCode; reason?: string };

/** 所有 policy 共用的基础字段（P2-1：invalidId / guards / write 推广到全部类型）。 */
interface PolicyBase {
  invalidId?: Denial; // 路径 / 查询标识非法时的现状码（org 400、permission idParam 404、job/context.uuidParam 400 …）
  guards?: readonly GuardName[]; // 附加具名守卫（grants 的 other_settings 附加能力、person-links 固定拒绝、经理身份 …）
  write?: WritePolicy; // 写路由必填（platform / admin / self / relation / object 都可能写）
}

export type RoutePolicy =
  | (PolicyBase & { kind: 'public'; reason: string; dec: string }) // 不要求成员身份；令牌 / 业务资格由 guards 强制（DEC-291 Q2）
  | (PolicyBase & { kind: 'platform'; fields: FieldsPolicy }) // 平台运营身份；write 为平台命令（Idempotency-Key + If-Match，platform.ledger）
  | (PolicyBase & { kind: 'member'; reason: string; fields: FieldsPolicy }) // 任一有效租户成员
  | (PolicyBase & { kind: 'admin'; capability: AdminCapability; alias?: string; scope?: ScopePolicy; fields: FieldsPolicy }) // alias = 现有 tenant.* 动作名；scope 供审计等能力级列表
  | (PolicyBase & { kind: 'self'; target?: Target; button?: ButtonPolicy; fields: FieldsPolicy }) // 绑定员工本人 + 叠加授权器 selfService
  | (PolicyBase & { kind: 'own'; predicate: PredicateName; target?: Target; locator?: LocatorName; denied?: Denial; fields: FieldsPolicy }) // 本人数据：列表用谓词；单条用 target + locator（序列同步任务）
  | (PolicyBase & { kind: 'relation'; relation: RelationName | Selector<RelationName>; target: Target; denied: Denial; rows?: RowsPolicy; fields: FieldsPolicy })
  | (PolicyBase & { kind: 'exception'; guard: GuardName; dec: string; fields: FieldsPolicy }) // 具名例外（DEC-135）
  | (PolicyBase & { kind: 'button'; object: ObjectSelector; button: Selector<{ code: string; level: ButtonLevel }>; denied: Denial }) // 独立按钮策略，供 any / all 组合（调动部门选择：任一发起按钮）
  | ObjectPolicy
  | (PolicyBase & { kind: 'any'; of: readonly RoutePolicy[] }) // OR：任一通过即通过；出口字段取通过分支的 fields；全失败按第一个分支的拒绝码
  | (PolicyBase & { kind: 'all'; of: readonly RoutePolicy[]; fields: FieldsPolicy }); // AND：全部通过；出口字段由组合层声明

interface ObjectPolicy extends PolicyBase {
  kind: 'object';
  object: ObjectSelector;
  operation: OperationSelector;
  button: ButtonPolicy;
  scope: ScopePolicy | { byObject: Readonly<Record<ObjectCode | '*', ScopePolicy>> };
  fields: FieldsPolicy;
  rows?: RowsPolicy;
  failureAudit?: FailureAuditPolicy; // 导入入口必填（§3.5）
}

export type ScopePolicy =
  | { mode: 'list'; predicate: PredicateName; view?: string } // 列表：谓词由框架注入 listQuery（§3.3）
  | { mode: 'point'; target: Target; locator: LocatorName; denied: Denial; view?: string }
  | { mode: 'see-all'; view?: string; creatorLocator?: LocatorName; denied: Denial | { status: 200; empty: true } } // 无归属对象（DEC-121）：看全部或创建人；view = 数据集（编制方案）；合同主数据列表无看全部时现状为 200 空列表
  | { mode: 'guard'; guard: GuardName; denied: Denial }
  | { mode: 'none'; reason: string };

export type Target =
  | { param: string } | { body: string } | { query: string }
  | { record: LocatorName; attribute: string }
  | { derived: GuardName };

export interface WritePolicy {
  fields: FieldsFrom;
  controls?: readonly string[];
  commandOnly?: boolean;
  footprint: GuardName | { none: true; reason: string }; // 提交前足迹守卫（§3.2 ②）；不再有裸 generic
  result: GuardName | { generic: { targets: string; locator: LocatorName } } | { none: true; reason: string }; // 返回后守卫（§3.2 ③）
}
type FieldsFrom =
  | 'body' | 'body.fields+customFields' | 'contracts.fields+customFields' | 'body+subdivisions'
  | { rows: string } | { guard: GuardName } | { none: true; reason: string };

export interface RowsPolicy {
  path: string; // 'rows[*]' | 'items[*]' | 'items[*].command'
  operation?: OperationSelector; // 逐行操作（含分支清单）
  button?: ButtonPolicy;
  relation?: Selector<RelationName>; // 逐行关系（合同待办批量：action → 当前审批人 / 发起人）
  fields: FieldsFrom;
  target?: Target;
  batch: 'atomic' | 'receipt';
}

export type FieldsPolicy =
  | { mode: 'shape'; shape: ShapeName }
  | { mode: 'projector'; projector: ProjectorName; shape: ShapeName; format?: 'text/csv' }
  | { mode: 'fixed'; keys: readonly string[]; dec: string; intersect?: ObjectCode } // 固定键；intersect = 再 ∩ 该对象字段权（经理员工列表）
  | { mode: 'all'; dec: string } // 整对象不裁剪（DEC-102 流程管理员分支）
  | { mode: 'none'; reason: string };

export interface FailureAuditPolicy {
  kind: 'import';
  rows: 'rows' | 'items';
  objectType: Selector<string>;
  anchors: MapperName;
  resolveAnchors?: GuardName;
  scopeEmployee?: Target;
}
```

### 2.3 缺失即失败 v3（P2-2 残留：中间件按最终路径登记；P2-10：HEAD）

**注册与绑定**

- `declare(router, method, path, policy, handler)`：为每次调用生成独立包装 `w`，登记 `Declaration = { method, localPath, policy, router, seen: false }`；同一 `(router 实例, method, localPath)` 重复声明抛错；多方法用 `declareEach`。
- `mount(router, prefix, module)`：调用 `router.route(prefix, module)` 前后对比 `router.routes`，把新增条目的**最终路径**（Hono `mergePath` 规范化后）回填到 `module` 上每条声明与中间件登记的 `expectedFullPath`；嵌套挂载逐层回填。不自行拼接字符串，只读 Hono 的结果。
- `useMiddleware(router, path, mw)`：调用 `router.use(path, mw)` 后读取 `router.routes.at(-1).path` 作为该中间件的登记路径（Hono 已规范化：`'*'` → `/*`），再登记进 `middlewareSet`；经 `mount` 时按上条回填最终路径。实测正例：顶层 `use('*')` → `ALL /*`；子路由器 `use('*')` 挂在 `/api` → `ALL /api/*`；两层嵌套 `inner.use('*')` 挂在 `/api` 下的 `/sub` → `ALL /api/sub/*`；`use('/api/tenant/*')` 不变。现有 7 条中间件改为经 `useMiddleware` 登记，三条源码里的 `'*'` 以 `/*` 入表。

**验证算法（`verifyRouteDeclarations(app, { baseline, exceptions })`，`createApp` 末尾调用）**

1. 遍历 `app.routes`；`fn = r.handler`，循环 `fn = fn['__COMPOSED_HANDLER']` 直到不存在。
2. `fn ∈ middlewareSet` → 要求 `r.method === 'ALL'` 且 `r.path === 登记的最终路径`；`fn ∈ declaredByHandler` → 要求 `r.method === d.method && r.path === d.expectedFullPath && !d.seen`，否则 `ROUTE_ALIAS`；其余 → 未声明：在 `baseline ∪ exceptions` 内 → warn，否则抛错（`.all()` 路由方法为 `ALL` 且不在 `middlewareSet` → 未声明）。
3. 收尾：每条声明都必须 `seen`；`baseline ∪ exceptions` 每条都在运行时出现且未声明。
4. **封闭**：`app.router.match('GET', '/__route_policy_seal__')` 预构建匹配器（此后 Hono 自身拒绝 `add`，实测），再 `Object.freeze(app.routes)`。
5. 返回运行时清单。

**包装的运行时自检（P2-10）**：按**有效匹配方法**核对：`effective = c.req.method === 'HEAD' ? 'GET' : c.req.method`，要求 `effective === d.method && c.req.routePath === d.expectedFullPath`，否则 500 `ROUTE_POLICY_MISMATCH`。HEAD 请求沿用 GET 声明的全部权限检查与投影，响应体由 Hono 在 dispatch 层去除（实测 HEAD 命中 GET 路由返回 200、空体），涉及 107 条 GET 的隐式 HEAD；`/healthz` 的 HEAD 同样 200。

### 3.3 列表 / 本人数据 / 审计的查询契约 v3（P2-3：框架组装与执行）

**契约**：列表、计数与聚合一律通过 `access.listQuery(tx, spec)` / `access.count(tx, spec)` / `access.aggregate(tx, spec)` 由框架**组装并执行**；处理函数不再拿到谓词 SQL。

```ts
interface QuerySpec {
  columns: SQL;                                   // 列表达式；可含标量子查询（不改变行集合）
  from: { table: string; alias: string };         // 只能是表标识符
  joins?: readonly Join[];                        // INNER / LEFT JOIN 表标识符，ON 为谓词
  lookups?: readonly Lookup[];                    // LEFT JOIN LATERAL：模块给出子查询主体与 ORDER BY，框架追加 LIMIT 1（最新快照 / 最新版本一类）
  with?: readonly Cte[];                          // CTE：kind 'rows'（参与行集合，受关键字禁令）或 'lookup'（只能 LEFT JOIN 按键关联，允许 GROUP BY）
  where?: readonly SQL[];                         // 业务过滤谓词；可含 EXISTS / 标量子查询（含 LIMIT 1）
  scopeColumns: { org?: SQL; person?: SQL; creator?: SQL; employee?: SQL; department?: SQL }; // 供具名谓词绑定列
  orderBy: SQL;
  page?: { limit: number; offset: number };       // 只能由框架加 LIMIT / OFFSET
  groupBy?: SQL;                                  // 只能由框架放在 WHERE 之后（aggregate）
}
```

- **组装顺序固定**：`WITH … SELECT columns FROM from JOIN … LEFT JOIN LATERAL (… LIMIT 1) … WHERE <具名范围谓词> AND <where…> [GROUP BY] ORDER BY orderBy [LIMIT OFFSET]`，范围谓词永远在 WHERE 首项、LIMIT / OFFSET / GROUP BY 之前。
- **行集合部分的关键字禁令**：`from`、`joins`、`with(kind: 'rows')`、`lookups` 主体中出现 `LIMIT / OFFSET / FETCH / GROUP BY / HAVING / DISTINCT / UNION / INTERSECT / EXCEPT / OVER(` → 组装时抛 `ROUTE_POLICY_BAD_SOURCE`（扫描模块提供的 SQL 文本，字符串字面量除外）；`columns`、`where`、`with(kind: 'lookup')` 不受禁令（标量子查询与 EXISTS 不改变外层行集合；lookup CTE 只能 LEFT JOIN）。禁止模块传入已分页 / 已聚合的 source。
- **执行归框架**：`listQuery` 内部 `tx.execute(composed)`，并记录 `access.queriesExecuted += 1`。`scope.mode: 'list'` 与 `kind: 'own'` 的路由在处理函数返回时 `queriesExecuted === 0` → 500 `ROUTE_POLICY_UNSCOPED`（测试环境抛错）。谓词 SQL 不再暴露给处理函数，"取出即算消费"的假通过路径不存在。
- **lint 范围扩大**：`apps/api/src/modules/**/*.ts`、`apps/api/src/audit/**/*.ts`（含 `queries.ts`、`read-model.ts`、`lists.ts` 等查询辅助文件）禁止参数化分页 `LIMIT ${…}` / `OFFSET ${…}` 与 Drizzle `.limit(` / `.offset(`（常量 `LIMIT 1` 标量 / lateral 查找允许，由框架的 `lookups` 生成或在标量子查询内）；`route-policy/` 目录除外。
- **与现有查询的关系**：现有 `listOrgSnapshot / listJobObjects / listEmployees / listContracts / listTodos / auditViewer` 等改为返回 `QuerySpec`（或其 `from / joins / where / columns` 片段），分页参数由路由传给框架；`employmentVisibilitySql`、`instanceScopeSql`、`personScope`、`audit.visibility` 等成为具名谓词，由框架按 `scopeColumns` 绑定。审计五段逻辑的落点不变（v2 §3.3 表），但"分页前过滤"改为由组装顺序保证。
- **FW-03 / FW-01 测试计划（三条内存 SQL 反例 + 正例）**：

| 反例 | 期望 |
|---|---|
| 处理函数不调用 `listQuery`，自行 `tx.execute` 全量查询并返回 | 返回守卫 `ROUTE_POLICY_UNSCOPED`（测试抛错）；lint 同时报参数化分页 |
| `from` / `with(rows)` 含 `LIMIT 1` 的子查询（source 先分页） | 组装时 `ROUTE_POLICY_BAD_SOURCE` |
| 正确：先范围过滤再分页 | inside 一行；outside 样本排序在前时仍不出现；`count` = inside 条数 |
| lookup：`LEFT JOIN LATERAL (… ORDER BY version_no DESC LIMIT 1)` 由框架生成 | 允许；行数不变 |

### 4.5 断言方式 v3（P2-7 ③④）

- **校验和四组**：
  - 业务：全部带 `tenant_id` 的 RLS 表，**含 `permission_*`**（作为"权限管理业务"子组单独报告）；

  - 审计与出站：`audit_events / audit_operation_logs / audit_command_failures / audit_object_creators` 与全部 `*_outbox*` 表。

- `receipt`：台账 +（实际成功记录的行数）——合同待办批量按 `key:index` 逐行 runCommand，失败行不入台账；范围外 / 不可编辑行的对象快照前后一致。
- 状态码：断言 status 与 `error.code`（含 `details.reason`），与声明 `Denial` 逐项相等；未知字段预期 `403 FORBIDDEN`。

   - 运行时未声明集合（FW-01 产出的 JSON）`⊆ PR.baseline ∪ PR.exceptions`，且两者每条都在运行时出现并未声明。
   - 在 main 自身（push 到 main）上同样运行：`main.baseline` 只能相对上一提交减少（`git show HEAD~1:…` 比较），防止直接推送。
3. **例外条目**：`{ method, path, pr, owner, approvedBy: 'DEC-xxx', exitBatch: 'B' | 'C', dependsOn: ['#97'] }`；`OPEN_BATCHES` 常量只能由 PR-B / PR-C 的收尾提交修改（同样与 main 比较只减不增）。
4. **依赖 PR 关闭 / 取消**：`dependsOn` 的 PR 关闭时，责任人在下一次合并 main 的 PR 删除该例外；基线条目对应路由消失 → FW-01 陈旧条目检查红，由 PR-B 删除。PR-B 前置 PR 任一关闭不影响 PR-B 派发。
5. **清零与强制**：`baseline ∪ exceptions` 为空即 strict；退到 PR-C 时 `OPEN_BATCHES` 加 `'C'`，强制切换推迟到 PR-C 合并。
6. **强制生效后**：缺声明 → `createApp` 抛错；`declare` 是唯一注册入口。
7. **冲突处理**：每次合并 main 后在 PR 描述逐项列出基线 / 例外减项。

## 附录 B：运行时枚举脚本（FW-01 的前身；附录 A 的权威来源）

```ts
// tests/acceptance/_f039_enum.test.ts（临时，用后删除）；F039_OUT=… pnpm vitest run tests/acceptance/_f039_enum.test.ts
import { writeFileSync } from 'node:fs';
import { createApp } from '@italent/api';
import { it } from 'vitest';
const COMPOSED = '__COMPOSED_HANDLER';
function unwrap(handler: unknown) {
  let depth = 0; let fn = handler;
  while (fn && typeof fn === 'function' && (fn as Record<string, unknown>)[COMPOSED]) { fn = (fn as Record<string, unknown>)[COMPOSED]; depth += 1; }
  return { depth, fn };
}
it('enumerate routes', () => {
  // 任何数据库访问都抛错的 Proxy：注册阶段不得访问数据库
  const db = new Proxy({}, { get: (_t, p) => { if (typeof p === 'symbol' || p === 'then') return undefined; throw new Error(`db access: ${String(p)}`); } });
  const app = createApp({ db: db as never });
  const rows = app.routes.map((r) => ({ method: r.method, path: r.path, composedDepth: unwrap(r.handler).depth }));
  writeFileSync(process.env.F039_OUT!, JSON.stringify({ total: rows.length, endpoints: rows.filter((r) => r.method !== 'ALL').length, middleware: rows.filter((r) => r.method === 'ALL').length, rows }, null, 1));
});
```

结果：`total 234, endpoints 227, middleware 7`；子路由器端点 `composedDepth = 1`（父级 onError 包装），直挂端点 0。Hono 4.13.12。
