## 0. 一页摘要

- **声明（§2）**：每条路由改用 `declare(router, method, path, policy, handler)` 注册。policy 九类：`public / platform / member / admin / self / own / relation / exception / object`，可用 `any`（OR）/ `all`（AND）组合。`object` 类的对象、操作、按钮都可以是**选择器**（由路径参数、请求体字段、已加载记录或具名映射函数决定），批量 / 导入用 `rows` 逐行声明操作、按钮、字段与范围目标。范围外与标识非法都声明 **status + error.code**（DEC-291 Q1）。附录 A 对运行时枚举出的 **227 个端点**逐条给出确定映射，没有"迁移时定"。
- **执行点（§3）**：七步——最小安全解析 → 功能权限 → 写字段 → 范围 / 关系 → 处理函数（命令事务内足迹守卫）→ 返回后复核（首次与重放同路径）→ 出口投影与守卫。写入声明必填 `footprint`（事务内）与 `result`（返回后）两个具名守卫，现有任职 / 编制 / 职务 / 组织 / 合同 / 审批交接 / 人员序码的锁内复核与足迹检查原样收编。列表与审计走**强制查询契约**（`access.listQuery` / `access.scopePredicate`，未消费谓词即 500 fail-closed）。出口按**响应路径 × 对象 × 编码**投影（键、字段名数组、`{field,from,to}` 条目、`custom:<id>`、`preset:` 前缀、CSV 列），系统字段按查看权判断不整体豁免；未登记路径一律删除并告警（测试环境抛错）。四个导入入口的失败请求仍进 `withFailedImportLog`。
- **缺失即失败（§2.3）**：按**注册实例**绑定最终 method / path；`.on` 多方法逐个展开；同一处理函数的别名、未登记中间件、`.all` 路由、多层 `onError` 包装都能检出；验证后用 Hono 自身的"匹配器已构建"不变量封闭注册（实测 `Object.freeze(app.routes)` 挡不住晚注册，见 §2.3）。
- **CI 反向测试（§4）**：七个固定用户只是**模板**（A / H / N / O / F / B / P / S），按路由的权限分支派生场景用户；必测项由声明**自动推导**成"路由 × 权限分支 × 场景"矩阵，缺样本、零用例、未执行直接失败；负例一律用合法请求隔离单个权限；校验和分业务数据 / 命令台账 / 失败审计三组比对；列表用分页敏感样本证明过滤发生在分页与计数之前。
- **分步上线（§3.7、§8）**：PR-A = 框架 + 反向测试 + 迁移 **126** 个端点（platform、tenant-settings、permission routes / data-scope / scope-policy、job、establishment、personnel、contracts、self-service、`/healthz`）；**101** 个端点进精确 method / path 基线（org、employment 含 transfer、approval、audit、permission/user-routes）。CI 用集合子集比较；有限例外有责任人、来源 PR、退出批次（DEC-291 流程①）。基线与例外同时为空即切强制。

## 2. 设计一：路由权限声明 v2

### 2.1 类型（`apps/api/src/route-policy/types.ts`）

```ts
/** 选择器：静态值，或由请求某处决定。取值只用步骤 0 的最小安全解析结果；取不到或不在映射表 → 400 VALIDATION_FAILED，不进入鉴权。 */
type Selector<T> =
  | T
  | { from: 'param' | 'body' | 'query'; path: string; map: Readonly<Record<string, T>> }
  | { from: 'mapper'; mapper: MapperName }; // 纯函数映射（如 contracts.commandButton = contractAction(operation, mode)），须登记测试

type ObjectSelector = Selector<ObjectCode> | { from: 'record'; locator: LocatorName; attribute: 'fieldObjectCode' };
type OperationSelector = Selector<'view' | 'create' | 'update' | 'delete'>;
type ButtonPolicy = Selector<{ code: string; level: ButtonLevel }> | { none: true; reason: string }; // 无按钮必须写原因（P3-4）

/** 拒绝码：范围外 / 关系不成立 / 标识非法都同时声明 status 与 error.code（DEC-291 Q1），含 details.reason（如 APPROVAL_SCOPE_DENIED）。 */
type Denial = { status: 403 | 404; code: ErrorCode; reason?: string };

export type RoutePolicy =
  | { kind: 'public'; reason: string; dec: string; guards: readonly GuardName[] } // 不要求成员身份；令牌 / 业务资格由守卫强制（DEC-291 Q2）
  | { kind: 'platform' }
  | { kind: 'member'; reason: string; fields: FieldsPolicy } // 任一有效租户成员（本人菜单、本人有效权限）
  | { kind: 'admin'; capability: AdminCapability; alias?: string; fields: FieldsPolicy; write?: WritePolicy } // alias = 现有 tenant.* 动作名，框架按 MODULE_ACTIONS 反查，保持判定不变
  | { kind: 'self'; target?: Target; button?: ButtonPolicy; guards?: readonly GuardName[]; fields: FieldsPolicy; write?: WritePolicy } // 绑定员工本人 + 叠加授权器 selfService
  | { kind: 'own'; predicate: PredicateName; fields: FieldsPolicy } // 本人数据：具名 SQL 谓词 + 强制消费
  | { kind: 'relation'; relation: RelationName; target: Target; denied: Denial; fields: FieldsPolicy; write?: WritePolicy; guards?: readonly GuardName[] }
  | { kind: 'exception'; guard: GuardName; dec: string; fields: FieldsPolicy } // 具名例外（DEC-135）
  | ObjectPolicy
  | { kind: 'any'; of: readonly RoutePolicy[] } // OR：任一通过即通过；全失败按第一个分支的拒绝码
  | { kind: 'all'; of: readonly RoutePolicy[] }; // AND

interface ObjectPolicy {
  kind: 'object';
  object: ObjectSelector;
  operation: OperationSelector;
  button: ButtonPolicy;
  scope: ScopePolicy | { byObject: Readonly<Record<ObjectCode, ScopePolicy>> }; // 范围可随选中对象变化（职务：positions 按 orgId，其余只认看全部 / 创建人）
  fields: FieldsPolicy;
  write?: WritePolicy; // operation ≠ view 必填
  rows?: RowsPolicy; // 批量 / 导入
  guards?: readonly GuardName[];
  failureAudit?: FailureAuditPolicy; // 导入入口必填（§3.5）
  invalidId: Denial | { status: 400; code: 'VALIDATION_FAILED' }; // 标识非法的现状码（org 400 / permission http.idParam 404 …）
}

export type ScopePolicy =
  | { mode: 'list'; predicate: PredicateName; view?: string } // 列表：谓词由框架提供并强制消费（§3.3）
  | { mode: 'point'; target: Target; locator: LocatorName; denied: Denial; view?: string } // 单对象：定位器取归属 → scopeAllowsInTransaction
  | { mode: 'see-all'; creatorLocator?: LocatorName; denied: Denial } // 无归属对象（DEC-121）：看全部或创建人
  | { mode: 'guard'; guard: GuardName; denied: Denial } // 事务内复杂规则（DEC-177 / 193 等）
  | { mode: 'none'; reason: string }; // 配置对象 / 平台 / 枚举字典 / 全租户例外，必须写原因

export type Target =
  | { param: string }
  | { body: string } // JSON 路径，支持 rows[*].orgId、parents.*.parentId
  | { query: string }
  | { record: LocatorName; attribute: string } // 先按路径 id 加载记录再取归属（合同 createdBy、人员申请 employeeId）
  | { derived: GuardName }; // 归属由业务计算得出（继承预览的部门），守卫在处理函数前计算

export interface WritePolicy {
  fields: FieldsFrom;
  controls?: readonly string[]; // 不算字段的控制键
  commandOnly?: boolean; // 只带命令不带字段的按钮（Employment.Submit / Withdraw / Revoke / RetryActivation、合同 withdraw）
  footprint: GuardName | 'generic'; // 事务内（行锁后）足迹复核（§3.2）
  result: GuardName | 'generic' | { none: true; reason: string }; // 首次 / 重放返回后复核（§3.2）
  resultTargets?: string; // generic 守卫复核的响应路径（'id' | 'results[*].orgId' | 'items[*].id' | 'instanceId'）
}
type FieldsFrom =
  | 'body' | 'body.fields+customFields' | 'contracts.fields+customFields' | 'body+subdivisions'
  | { rows: string } | { guard: GuardName } | { none: true; reason: string };

export interface RowsPolicy {
  path: string; // 'rows[*]' | 'items[*]' | 'items[*].command'
  operation: OperationSelector; // 逐行：由 objectId 有无 / operation / mode 决定
  button?: ButtonPolicy;
  fields: FieldsFrom;
  target?: Target; // 逐行范围目标
  batch: 'atomic' | 'receipt'; // AGENTS §10「批量」
}

export type FieldsPolicy =
  | { mode: 'shape'; shape: ShapeName } // 框架按形状投影（§3.4）
  | { mode: 'projector'; projector: ProjectorName; shape: ShapeName; format?: 'text/csv' } // 模块投影器 + 形状（出口守卫按形状核对）
  | { mode: 'fixed'; keys: readonly string[]; dec: string } // 固定最小键集合（DEC-135、DEC-057 最小字段）
  | { mode: 'none'; reason: string }; // 无业务字段（204、计数、枚举字典、平台 DTO）

export interface FailureAuditPolicy {
  kind: 'import';
  rows: 'rows' | 'items';
  objectType: Selector<string>;
  anchors: MapperName; // 原始行 → { orgId? employeeId? objectId? }（只取 UUID）
  resolveAnchors?: GuardName; // 任职：失败时在写日志事务内回查
  scopeEmployee?: Target;
}
```

### 2.3 缺失即失败 v2（P2-2）

**注册与绑定**

- `declare(router, method, path, policy, handler)`：为每次调用生成独立包装 `w`（即使同一业务 `handler` 被声明两次也是两个 `w`），登记 `Declaration = { method, localPath, policy, router, seen: false }` 到 `declaredByHandler: WeakMap<Function, Declaration>` 与 `declarationsByRouter: WeakMap<Hono, Declaration[]>`；同一 `(router 实例, method, localPath)` 重复声明抛错（只在同一路由器实例内去重，employment / contracts 的 `GET /settings`、`GET /records/:id` 分属不同子路由器，不误判）。`declare` 只接受单个 method；多方法用 `declareEach(router, ['GET','POST'], path, policy, handler)` 展开成多条声明。
- `mount(router, prefix, module)`：记录 `module` 上全部声明的 `expectedFullPath = mergePath(prefix, localPath)`（嵌套挂载逐层拼接），再调用 `router.route(prefix, module)`。模块内禁止直接 `router.route`（lint）。
- `useMiddleware(router, path, mw)`：把 `mw` 登记进 `middlewareSet`（含 path），再 `router.use(path, mw)`。现有 7 条中间件改为经此登记。

**验证算法（`verifyRouteDeclarations(app, { baseline, exceptions })`，`createApp` 末尾调用）**

1. 遍历 `app.routes` 每条 `r`；`fn = r.handler`，循环 `fn = fn['__COMPOSED_HANDLER']` 直到不存在（多层 `onError` 递归还原，实测两层嵌套 depth=2 可还原）。
2. `fn ∈ middlewareSet` → 要求 `r.method === 'ALL'` 且路径与登记一致，否则抛错；
   `fn ∈ declaredByHandler` → 取 `d`：要求 `r.method === d.method && r.path === d.expectedFullPath && !d.seen`，然后 `d.seen = true`；不满足即 `ROUTE_ALIAS`（同一包装出现在别的 method / path，或出现两次）；
   其余 → 未声明：`key = \`${r.method} ${r.path}\``；在 `baseline ∪ exceptions` 内 → `console.warn`；否则抛错。`.all()` 注册的路由方法为 `ALL` 且不在 `middlewareSet` → 按未声明处理，不再被"跳过 ALL"放过。
3. 收尾：每条声明都必须 `seen`（声明了但没挂进 `app` 的子路由器 → 抛错）；`baseline ∪ exceptions` 的每条都必须在运行时出现且未声明（陈旧条目 → 抛错）。
4. **封闭**：`app.router.match('GET', '/__route_policy_seal__')` 预构建匹配器——此后任何 `app.get/on/use/route` 都被 Hono 自身以 `Can not add a route since the matcher is already built` 拒绝（实测），再 `Object.freeze(app.routes)` 冻结清单。实测说明：只冻结 `routes` 不够——Hono `#addRoute` 先 `router.add` 再 `routes.push`，冻结只会让 push 抛错而路由已生效（实测晚注册路由返回 200）；预构建匹配器后晚注册返回 404。子路由器在 `route()` 之后再注册的路由不会进入父级（实测 404），因此封闭 `app` 即封闭全部。
5. 返回运行时清单（method、fullPath、policy 摘要）供 FW-01 与 PR 描述统计。

**包装的运行时自检**：`w` 执行时核对 `c.req.method === d.method && c.req.routePath === d.expectedFullPath`（实测子路由器内 `routePath` 为拼接后的完整路径），不一致返回 500 `ROUTE_POLICY_MISMATCH`（fail-closed），作为别名检出的第二道。

**FW-01 测试计划（用内存 Hono + 框架构造反例）**

| 反例 | 期望 |
|---|---|
| `.all('/x', h)` 未声明 | verify 抛错，且请求 POST /x 不会成功 |
| 同一包装 `w` 再 `router.post('/alias', w)` | `ROUTE_ALIAS` |
| verify 之后 `app.get('/late', …)` | Hono 抛 "matcher is already built"，GET /late 404 |
| 两层 `onError`（子路由器再挂子路由器） | depth=2 还原到原包装，正常匹配 |
| `declareEach(['GET','POST'])` 只声明 GET 后手工 `router.post` | POST 条目未声明 → 抛错 |
| employment 与 contracts 各有 `GET /settings` | 完整路径不同，均通过；同一子路由器重复声明 → 抛错 |
| 子路由器挂载后再 `module.get('/b')` | 父级无此路由（404），不计入清单（文档化） |
| 直接 `router.use(path, mw)` 未经 `useMiddleware` | "未登记中间件" 抛错 |
| 基线里残留已声明 / 已删除的条目 | 抛错 |

### 2.4 配置类与特例的判定保真

- 配置对象（`EmploymentSettings`、`EmploymentCustomField`、`ContractSettings`、`ContractRenewalRule`）声明为 `kind: 'object'`，功能权限仍走 `requirePermission(object.<op>)`，由 `authorizer.ts` 的 CONFIG_OBJECTS 分支判定（任职配置 = `other_settings`；合同配置 = tenant_admin 或 `contractConfiguration`）。框架**不**另加能力要求；任职 / 调动配置路由现有的 `tenant.employment.configuration.write` 别名用 `all([admin(other_settings, alias), object(update)])` 原样保留（两者在授权器里同义，声明只是显式化）。
- `/permission/admin-menus`、`/permission/me/objects/:objectCode`：`kind: 'member'`，普通成员得空列表 / 空权限，不改为管理员能力。
- `/org/person-candidates`：`kind: 'exception'`，守卫 `org.anyPersonFieldEditable`（任一 ORG_PERSON_FIELDS 可写即可），范围"全租户"（DEC-135），字段固定最小集合。
- 平台路由：`kind: 'platform'`；平台 DTO 没有字段权限目录，`fields: none`。
- 公开路由（DEC-291 Q2）：`kind: 'public'` 必须带 `guards`（令牌 / 资格守卫）；`/healthz` 的 `guards: []` 须写明 `reason: '健康检查无数据'`。FW-01 输出公开路由清单供每轮审查核对。

## 3. 设计二：框架执行点 v2

### 3.1 一次请求的执行顺序（`route-policy/enforce.ts`）

```
tenantContext / platformContext（现有中间件，经 useMiddleware 登记）
→ captureCommandFailures（现有）
→ [0] 最小安全解析：路径参数格式与 UUID 规范化（DEC-194，复用 F-017 口径）、枚举参数、请求体是否 JSON 对象、
      选择器用到的值、rows 数组与上限、If-Match / Idempotency-Key 格式；不合法按声明的 invalidId 或 400 VALIDATION_FAILED
      （C2 修订：取不出对象 / 模式 / 目标就无法鉴权，所以这些格式错误先于鉴权；其余业务校验仍在鉴权之后）
→ [1] 上下文：tenantOf(c)、expectedRevision
→ [2] 功能权限：按解析出的 object / operation / button 调 requirePermission；admin 按能力（或 alias 动作）；any / all 组合
→ [3] 写字段（pre）：按 write.fields 提取 → requireObjectWrite（任一字段不可写或未登记 → 403；显式 null 同样算写）
→ [4] 范围 / 关系（pre）：list → 准备具名谓词；point → 定位器 + scopeAllowsInTransaction；see-all；guard；relation；self 绑定
→ [5] 处理函数(c, access)：业务校验（zod）、access.listQuery / access.command；命令事务内由框架调用 write.footprint 守卫
→ [6] 返回后复核：首次与重放同一路径，重跑 [4] 的目标检查并调用 write.result 守卫
→ [7] 出口：按 fields 投影（形状 / 投影器 / 固定键 / 无）→ 出口守卫深扫 → CSV 经 access.csv 序列化
导入入口：[0] 先构造失败审计任务，[1]～[6] 整体包在 withFailedImportLog 内（§3.5）

- 响应码变化只发生在"无权用户发坏请求体"这一类：以前部分路由先 400 后 403，现在 [2] 先于业务 zod 校验 → 403 / 404。[0] 的格式错误仍为 400 / 声明的 invalidId，与现状一致。迁移时逐条核对现有 AC 测试的断言码并在 PR 描述列出。
- `access` 对象：`{ ctx, expectedRevision, selected: { object, operation, button }, scope?, viewableFields(object), editableFields(object), scopePredicate(columns), listQuery(tx, spec), command(c, spec), csv(spec), assertInScope(tx, target), assertLinked(tx, target) }`。

### 3.3 列表 / 本人数据 / 审计的强制查询契约（P2-3）

- `access.scopePredicate(columns)`：返回具名谓词 SQL（默认 `module.scopeSql`；任职 `employment.visibility` = DEC-177 两段 OR；审批 `approval.adminScope` / `approval.recipient` / `approval.initiatedOrParticipated`；人员 `personnel.personScope`；合同 `contracts.employeeVisibility`；编制通知 `establishment.notificationRecipient`；经理 `transfer.managerTeam`；审计 `audit.visibility`），并把 `access.scopeConsumed = true`。
- `access.listQuery(tx, { source, filters, order, page, count? })`：框架拼 `SELECT … FROM (source) s WHERE <predicate> AND <filters> ORDER BY … LIMIT/OFFSET`，`count` 用同一 WHERE；`source` 是模块给出的子查询 / CTE，须暴露谓词需要的列。
- **强制**：`scope.mode: 'list'` 与 `kind: 'own'` 的路由，处理函数返回时 `scopeConsumed` 为 false → 500 `ROUTE_POLICY_UNSCOPED`（测试环境抛错）。配套 lint：`modules/**/routes*.ts` 内禁止出现 `LIMIT` / `.limit(` / `OFFSET`（分页只能由框架在 `access.listQuery` 里加），避免"消费了谓词却没放进分页之前"。
- **验证**：FW-03 用分页敏感样本（范围外样本按排序必在第一页第一条）、`pageSize=1` 与 `count` 断言，证明过滤在分页与计数之前（§4.4）。
- **各段逻辑落点**：

| 逻辑 | 现状 | v2 落点 |
|---|---|---|
| own 的 userId 过滤（待办 / 通知 / 我发起 / 经理待办 / 编制通知接收人 / 序列同步回执） | queries.ts 的 SQL | 具名谓词 `approval.recipient` 等，`listQuery` WHERE |

### 3.4 出口投影契约（P2-5）

**形状登记**：`registerShape(name, { paths: [{ path, object | 'protocol' | 'passthrough', encoding, envelope?, derivedFrom? }] })`。

| 编码 | 含义 | 实例 |
|---|---|---|
| `keys` | 对象键 = 字段编码，按查看权保留 | 绝大多数 DTO、`fields` 子对象 |
| `customFieldKeys` | 键是自定义字段 id → 按 `custom:<id>` 判断 | 任职 / 合同 / 自助 `customFields` |
| `prefixedKeys` | 键 `preset:<f>` 按 `<f>`、`custom:<id>` 按 `custom:<id>` | 调动表单 `fieldModes`、自助预览 |
| `fieldNameValues` | 数组元素是字段名（可带 `preset:` 前缀） | 任职 `skipped[*].fields`、`warnings[*].fields`、`completion-todos[*].fieldCodes`、`excludedAutofillFields`、审批日志字段名数组、`transfers/catalog` 的 `viewableFields / editableFields` |
| `fieldEntries` | 元素对象里 `field` 键是字段名，其余（from / to / value）是值 | 任职 `changes[*].fields[*]`、审计 `changes[*]` |
| `derivedText` | 文本由已投影结构派生，声明 `derivedFrom` 路径；投影器测试须断言隐藏字段标签不出现 | 审计 `content` |
| `csvColumns` | 列名 → 字段；经 `access.csv` 过滤后序列化 | 编制复制报告、合同导入错误报告 |

**投影与守卫算法**：按形状遍历响应；每个对象节点的键须 ∈ `catalog(object)`（按查看人 `viewableFields` 保留，**系统字段同样按查看权**，不整体豁免）∪ 该路径的 `envelope`；其余键与**未被任何路径覆盖的对象节点**：测试环境抛 `ROUTE_POLICY_UNDECLARED_KEY`，生产删除并写 `route_policy_leak` 日志（DEC-291 Q3）。同名字段按路径所绑定的对象判断，不做禁止集合并集。`passthrough` 路径（`page`、`window`、`nextCursor` 等纯协议结构）必须写原因。

### 3.5 失败导入日志（P2-6）

四个导入入口（`POST /org/import`、`POST /job/import`、`POST /employment/employees/:id/import`、`POST /contracts/imports`）声明 `failureAudit`。调用顺序：

1. 步骤 [0]：用 `rawImportRows(body, rows)` 取行数与 UUID 归属（只取可识别编号，不存其他输入值），按 `objectType` 选择器与 `anchors` 映射构造任务 `{ tenantId, userId, commandId, objectType, total, anchors | resolveAnchors, scopeEmployeeId? }`；职务导入 `kind` 不合法时 `objectType = 'job'`、`total = 0`（现状）。
2. 框架以 `withFailedImportLog(db, task, () => 步骤 [1]～[6])` 包住；按钮 / 字段 / 范围 / 格式失败都在包内 → 任务级失败日志（`successCount 0`、`failureCount total`、逐行归属、错误报告按 `failedImportReport` 取 details.errors 或 zod issues 的行号）；`total = 0` 与结果未知不记（现状）。
3. 预览 / 错误报告下载（`/imports/preview`、`/imports/errors`、`/import/forward-update-preview`）现状不包装，v2 保持；错误报告下载在处理函数内记 `download` 操作日志不变。
4. 等价性：现状职务 / 组织导入先 `requireNew` 再包装，字段 / 按钮 / 范围检查在包内；v2 全部在包内，差异只在 `requireNew`（REVISION_CONFLICT）也进包——这是多记一条失败日志，不是少记；写进迁移 PR 描述。

### 4.5 断言方式（P2-7 ⑤～⑧）

- 合法请求：负例用 `example` 或样本 GET，携带合法 If-Match 与新 Idempotency-Key，只撤掉被测的那一项权限；不再用 `{}`。
- 状态码：断言 status 与 `error.code`（含 `details.reason`），与声明的 `Denial` 逐项相等。
- 校验和三组：**业务**（全部带 `tenant_id` 的 RLS 表，排除 `permission_*`、`command_ledger`、`audit_*`、`*_outbox*` 的 md5 + 行数）、**台账**（`command_ledger` 行数）、**审计**（`audit_events / audit_operation_logs / audit_command_failures` 行数与最后一条）。负例：业务、台账不变；审计只允许追加失败类记录（`audit_command_failures` ≤ 1 条、导入失败日志 ≤ 1 条），其他审计表不变。GET：三组全不变。`receipt`：成功行允许业务变化，范围外对象的快照（H 读取）前后一致，台账 +1。权限变更（撤权）在比对窗口之外完成。
- 深扫：`walk(json)` 收集全部键路径与值；按形状编码解释字段名出现在值里的情况。
- 分页敏感：范围外样本的排序键设为最小 / 最大，`pageSize = 1` 请求首页，断言首条是范围内样本；有 `count / total` 的断言数量。

## 8. "告警 + 白名单" → 强制：切换条件与收敛计划（P2-9）

1. **基线冻结**：PR-A 合并时把运行时未声明的 101 个端点以精确 `{ method, path, module, batch: 'B' }` 写入 `allowlist.baseline.json`；文件只允许删条目或整文件删除（CODEOWNERS 指向审查合并窗口）。FW-01 断言 `运行时未声明集合 ⊆ baseline ∪ exceptions` 且两者每条都在运行时出现并未声明。
2. **有限例外**（DEC-291 流程①）：`allowlist.exceptions.json` 条目 `{ method, path, pr, owner, exitBatch: 'B' | 'C', approvedBy: 'DEC-xxx', dependsOn: ['#97'] }`。CI 核对：字段齐全、`pr` 为数字、`exitBatch` ∈ 开放批次常量 `OPEN_BATCHES`、同一端点不重复、总数 ≤ `MAX_EXCEPTIONS`（随 DEC 登记）。登记例外的 PR 必须同时在描述里写明责任人与退出条件。
3. **只减不增**：自 PR-A 合并起，基线不得增加；例外只能由审查合并窗口登记（DEC-291 Q5）；新模块与新派发任务的路由一律直接声明。
4. **依赖 PR 关闭 / 取消**（DEC-291 Q7）：例外 `dependsOn` 的 PR 关闭或取消时，由责任人在下一次合并 main 的 PR 里删除该例外；基线条目对应的路由若因 PR 关闭而不再存在，FW-01 的"陈旧条目"检查会红，由 PR-B 删除。PR-B 的前置 PR（#83 / #96 / #97 / #99）任一关闭，不影响 PR-B 派发时点：其余合并后即派。
5. **清零与强制**：`verifyRouteDeclarations` 在 `baseline ∪ exceptions` 为空时自动 strict。PR-B 迁移 org / employment / transfer / approval / audit / user-routes 并删除基线文件；若 #106 / #107 退到 PR-C（DEC-291 Q6），强制切换推迟到 PR-C 合并，`OPEN_BATCHES` 加 `'C'`，例外 `exitBatch: 'C'`。
6. **强制生效后**：缺声明 → `createApp` 抛错 → 全部测试红；`declare` 是唯一注册入口（`router.get` 直接注册被 FW-01 捕获，未经 `useMiddleware` 的中间件同样）。
7. **冲突处理**：每次合并 main 后，在 PR 描述逐项列出被动合并的路由文件、改码的现有测试、基线 / 例外的减项。

// tests/acceptance/_f039_enum.test.ts（临时，用后删除）；F039_OUT=… pnpm vitest run tests/acceptance/_f039_enum.test.ts
import { writeFileSync } from 'node:fs';
import { createApp } from '@italent/api';
import { it } from 'vitest';
const COMPOSED = '__COMPOSED_HANDLER';
function unwrap(handler: unknown) {
  let depth = 0; let fn = handler;
  while (fn && typeof fn === 'function' && (fn as Record<string, unknown>)[COMPOSED]) { fn = (fn as Record<string, unknown>)[COMPOSED]; depth += 1; }
  return { depth, fn };
}
it('enumerate routes', () => {
  // 任何数据库访问都抛错的 Proxy：注册阶段不得访问数据库
  const db = new Proxy({}, { get: (_t, p) => { if (typeof p === 'symbol' || p === 'then') return undefined; throw new Error(`db access: ${String(p)}`); } });
  const app = createApp({ db: db as never });
  const rows = app.routes.map((r) => ({ method: r.method, path: r.path, composedDepth: unwrap(r.handler).depth }));
  writeFileSync(process.env.F039_OUT!, JSON.stringify({ total: rows.length, endpoints: rows.filter((r) => r.method !== 'ALL').length, middleware: rows.filter((r) => r.method === 'ALL').length, rows }, null, 1));
});
```
